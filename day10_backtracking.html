    if is_target(state):
        result.append(state)

📝 2. [실습 문제 10] 피로도 (프로그래머스 Lv.2)

def solution(k, dungeons):
    n = len(dungeons)
    visited = [False] * n
    ans = 0
    
    def dfs(curr_k, count):
        nonlocal ans
        ans = max(ans, count)
        for i in range(n):
            req, consume = dungeons[i]
            if not visited[i] and curr_k >= req:
                visited[i] = True
                dfs(curr_k - consume, count + 1)
                visited[i] = False
                
    dfs(k, 0)
    return ans
