📌 1. BFS (Breadth-First Search) 핵심 특성

시작 노드에서 가까운 노드부터 레벨 순서대로 탐색하므로, **가중치가 동일(1)한 그래프에서 최단 경로**를 보장합니다.

from collections import deque

queue = deque([(start_r, start_c)])
dist[start_r][start_c] = 1

while queue:
    r, c = queue.popleft()
    for dr, dc in [(-1,0),(1,0),(0,-1),(0,1)]:
        nr, nc = r + dr, c + dc
        if valid(nr, nc) and dist[nr][nc] == 0:
            dist[nr][nc] = dist[r][c] + 1
            queue.append((nr, nc))

📝 2. [실습 문제 9] 게임 맵 최단거리 (프로그래머스 Lv.2)

def solution(maps):
    N, M = len(maps), len(maps[0])
    dist = [[0] * M for _ in range(N)]
    queue = deque([(0, 0)])
    dist[0][0] = 1
    
    while queue:
        r, c = queue.popleft()
        if r == N - 1 and c == M - 1:
            return dist[r][c]
        for dr, dc in [(-1,0),(1,0),(0,-1),(0,1)]:
            nr, nc = r + dr, c + dc
            if 0 <= nr < N and 0 <= nc < M:
                if maps[nr][nc] == 1 and dist[nr][nc] == 0:
                    dist[nr][nc] = dist[r][c] + 1
                    queue.append((nr, nc))
    return -1
